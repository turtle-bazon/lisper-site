Простейшее приложение на базе Hunchentoot
Автор: archimag - 2009-12-15T18:52:10.000000+03:00
(asdf:operate 'asdf:load-op 'hunchentoot)

(defun request-info (req)
  (hunchentoot:request-uri req))

(hunchentoot:start (make-instance 'hunchentoot:acceptor
                                  :request-dispatcher 'request-info
                                  :port 8080))

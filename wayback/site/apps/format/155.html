;; packages

(defmacro package-symbols (package &key symbol-type)
  (check-type symbol-type (member :external :internal))
  `(let (symbols)
     (with-package-iterator (iterator ,package ,symbol-type)
       (loop (multiple-value-bind (morep symbol) (iterator)
               (unless morep (return))
               (push symbol symbols))))
     symbols))

(defun package-external-symbols (package)
  (package-symbols package :symbol-type :external))

(defun package-internal-symbols (package)
  (package-symbols package :symbol-type :internal))

;; symbols

(defun type-of-symbol (symbol)
  (check-type symbol symbol)
  (cond ((macro-function symbol) :macro)
        ((fboundp symbol)        :function)
        ((boundp symbol)         :variable)
        ((find-class symbol nil) :class)
        (t                       :unknown)))

(defmacro emit-to-stream (stream &body body)
  `(labels ((->empty ()
              (terpri ,stream))
            (->text (format-string &rest args)
              (apply #'format ,stream format-string args)
              (->empty))
            (=text= (format-string &rest args)
              (let* ((string (apply #'format nil format-string args))
                     (=line= (make-string (length string) :initial-element #\=)))
                (format ,stream "~S~%" =line=)
                (format ,stream string)
                (format ,stream "~S~%" =line=)
                (->empty)))
            (-text- (format-string &rest args)
              (let* ((string (apply #'format nil format-string args))
                     (-line- (make-string (length string) :initial-element #\-)))
                (format ,stream string)
                (format ,stream "~S~%" -line-)
                (->empty)))
            (->doc (object type)
              (let ((documentation (documentation object type)))
                (when documentation
                  (format ,stream "~S~%" documentation)
                  (->empty)))))
     ,@body))

(defun emit-symbol (stream symbol)
  (emit-to-stream stream
    (->text "~A:~%" symbol)
    (case (type-of-symbol symbol)
      (:macro    (->doc symbol 'function))
      (:function (->doc symbol 'function))
      (:variable (->doc symbol 'variable))
      (:class    (->doc (find-class symbol) 'standard-class)))))

(defun emit-package (stream name)
  (emit-to-stream stream
    (=text= "Package: ~A~%" name)
    (->doc name 'package)
    (let ((external-symbols (package-external-symbols name))
          (internal-symbols (package-internal-symbols name)))
      (when external-symbols
        (-text- "External symbols:~%")
        (dolist (symbol external-symbols)
          (emit-symbol stream symbol)))
      (when internal-symbols
        (-text- "Internal symbols:~%")
        (dolist (symbol internal-symbols)
          (emit-symbol stream symbol))))))

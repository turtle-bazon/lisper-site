(with-input-from-string (s "#S(CONFIG :PARAM1 (r s) :PARAM2 2)")
  (let (c (read s))
    (format t "Readed: ~a~&" c)
    (format t "Param1: ~a~&" (config-param1 c))
    (format t "Param2: ~a~&" (config-param2 c))))

funcall inline example
Автор: dmitry_vk - 2010-09-11T21:32:11.000000+04:00
(declaim (inline foo))
(defun foo (x) (+ 1 x))

(defun bar (x)
  (funcall #'foo x))

(disassemble 'bar)

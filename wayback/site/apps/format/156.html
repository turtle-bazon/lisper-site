;; также лучше сделать не список консов (там у вас танцы с консами :) а список структур.
;; Соответственно будут два типа данных - структуры EVENT и их списки, ну и два типа методов -
;; для EVENT и для списка.

;; то что там с окружениями (stop в окружении, defun в defun) - всё это не хорошо. Я не совсем понял
;; механизма работы - но нужно заключать несколько определений функций в общий блок let - у них
;; будет разделяемое окружение.

(defpackage :file-event
  (:nicknames #:fevent)
  (:use       #:common-lisp)
  #+sbcl
  (:shadow    #:timeout)
  (:export    #:each
              #:wait
              #:do-file-event))

(in-package #:file-event)

;;; times

(defun sleep* (ms)
  (when (> ms 0)
    (sleep (/ ms 1000))))

(deftype timeout () '(integer 0 1000))

(defconstant +idle+ 0) ;; for time = 0

;;; EVENT structure

(declaim (inline %make-event))
(defstruct (event
            (:constructor %make-event))
  (time 0 :type timeout)
  (function nil :type function))

;; constructor

(declaim (ftype (function (timeout function) event) make-event))
(defun make-event (time function)
  (%make-event :time (+ (get-internal-real-time)
                        (/ (* internal-time-units-per-second time) 1000))
               :function function))

;; methods

(declaim (ftype (function (event) t) execute-event))
(defun execute-event (event)
  (sleep* (/ (- (event-time event)
                (get-internal-real-time))
             (/ internal-time-units-per-second 1000)))
  (funcall (event-function event)))

;;; EVENT-LIST

(declaim (ftype (function (timeout function) list) add-event each)
         (ftype (function () list) sort-event-list)
         (ftype (function (timeout function) function) closure-event)
         (ftype (function (stream function timeout) list) do-stream-readable-function
                                                          do-stream-closed-function))

(let (event-list stop-p)

  (declare (type list event-list) (type boolean stop-p))

  (defun add-event (time function)
    (push (make-event time function) event-list))

  (defun sort-event-list ()
    (setf event-list (sort event-list #'< :key #'car)))

  (defun closure-event (time function)
    (lambda ()
      (unless stop-p
        (add-event time function)
        (funcall function))))

  (defun each (time function)
    (add-event +idle+ (closure-event time function)))

  (defun wait ()
    ;; FIXME - какой-нибудь кошерный loop ?
    (do ()
        ((or stop-p (null event-list)))
      (sort-event-list)
      (handler-case
          (execute-event (car event-list))
        (t (c) (declare (ignore c)) nil))
      (setf event-list (cdr event-list))))

  ;;; STREAMs

  (defun do-stream-readable-function (stream function time)
    (each time
          (lambda ()
            (if (open-stream-p stream)
                (when (listen stream)
                  (funcall function))
                (setf stop-p t)))))

  (defun do-stream-closed-function (stream function time)
    (each time
          (lambda ()
            (unless (open-stream-p stream)
              (funcall function)
              (setf stop-p t))))))

(declaim (ftype (function (stream keyword function &optional timeout) list) do-file-event))
(defun do-file-event (stream event function &optional (time 10))
  (ecase event
    (:readable (do-stream-readable-function stream function time))
    (:closed   (do-stream-closed-function stream function time))))

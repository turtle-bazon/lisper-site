foldl
Автор: LinkFly - 2010-08-18T19:36:55.000000+04:00
(defmacro foldl (fn fst-arg (start end))
               `(loop with result = ,fst-arg
                   for i from ,start to ,end
                   do (setq result
                            (funcall ,fn ,fst-arg result i))
                   finally (return result)))

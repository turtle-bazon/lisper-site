cllib-find-system.lisp
Автор: dmitry_vk - 2009-11-02T22:48:17.000000+03:00
(defvar *cl-lib-path* (or *compile-file-pathname*
                          *load-pathname*))
;; для простоты предполагаем, что .asd лежат
;; рядом с текущим файлов

(defun cllib-find-system (system)
  (let* ((name (asdf::coerce-name system))
         (asd-path (merge-pathnames (make-pathname :name name
                                                   :type "asd"
                                                   :directory
                                                   (list :relative name))
                                    *cl-lib-path*)))
    (probe-file asd-path)))

(pushnew 'cllib-find-system asdf:*system-definition-search-functions*)

Парсим ODS

2015 , October 23
Tags: lisp ods

Внезапно потребовалось из программы на CL прочитать ODS файл. Подходящего решения не нашёл, правда особо не искал, потому что это довольно просто сделать самому:

  1. (ql:quickload "zip")
  2. (ql:quickload "cl-libxml2")
  3. (ql:quickload "parse-number")
  5. (defpackage #:my.tools.ods
  6.   (:use #:cl #:iter)
  7.   (:export #:parse-ods-file))
  10. (in-package #:my.tools.ods)
  12. (defun ods-xpath-map ()
  13.   (list* (list "office" "urn:oasis:names:tc:opendocument:xmlns:office:1.0")
  14.          (list "table" "urn:oasis:names:tc:opendocument:xmlns:table:1.0")
  15.          xpath:*default-ns-map*))
  17. (defun parse-ods-content (content)
  18.   (xtree:with-parse-document (doc content)
  19.     (let ((xpath:*default-ns-map* (ods-xpath-map)))
  20.       (iter (for sheet in (xpath:find-list doc "//table:table"))
  21.             (collect (parse-sheet sheet))))))
  23. (defun parse-sheet (sheet)
  24.   (list* (xpath:find-string sheet "@table:name")
  25.          (iter (for row in (xpath:find-list sheet "table:table-row"))
  28.                (collect (parse-row row)))))
  30. (defun parse-row (row)
  31.   (iter (for cell in (xpath:find-list row "table:table-cell"))
  32.         (let ((repeat-number (xpath:find-string cell "@table:number-columns-repeated")))
  33.           (dotimes (i (if repeat-number (parse-integer repeat-number) 1))
  34.             (collect (parse-cell cell))))))
  36. (defun parse-cell (cell)
  37.   (let ((value (xtree:text-content cell))
  38.         (value-type (xpath:find-string cell "@office:value-type")))
  39.     (cond
  40.       ((string= value-type "float")
  41.        (parse-number:parse-number value))
  42.       (t
  43.        value))))
  45. (defun parse-ods-file (pathname)
  46.   (zip:with-zipfile (zipfile pathname)
  47.     (parse-ods-content
  48.      (zip:zipfile-entry-contents
  49.       (zip:get-zipfile-entry "content.xml" zipfile)))))

Мне этого пока хватает.

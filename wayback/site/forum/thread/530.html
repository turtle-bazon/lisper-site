(defun im-to-texture (texture)
  (let ((png (png-read:read-png-file texture))
     (png-width (png-read:width png))
       (png-height (png-read:height png))
         (png-data  (make-array (list (* png-width png-height 1)))))
       (gl:bind-texture :texture-2d png)
                (gl:tex-image-2d :texture-2d
                         0
                         3
                         png-width
                         png-height
                         0
                         :RGB
                         :unsigned-byte
                         png-data)))

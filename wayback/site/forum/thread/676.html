У меня 7.1 Мб вот такой простой код
(defun csv-to-list (path &optional separator)
  (let (s result)
    (with-open-file
    (stream path
        :direction :input
        :if-does-not-exist :error)
      (loop for line = (read-line stream nil 'foo)
       until (eq line 'foo)
       do
       (progn
         (if separator
         (setf s (cl-ppcre:split  separator line))
         (setf s (cl-ppcre:split  ";" line)))
         (setf result (cons s result)))))
    (reverse result)))

Автор: Ballbreaker - 14.04.2010 13:39
(defun generate-power-set(source-set)
  (declare (optimize (debug 3)))
  (labels ((gen-power-set (set rest)
             (let ((results nil)
                   (next-el (car rest)))
               (break)
               (if (not next-el)
                   set
                   (progn
                     (dolist (elem set)
                       (push elem results)
                       (push (append elem (list next-el)) results))
                     (push (list next-el) results)
                     (gen-power-set results (cdr rest)))))))
    (gen-power-set '() source-set)))

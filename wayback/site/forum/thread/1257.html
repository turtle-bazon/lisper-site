motopeh - 29.10.2015 00:52, Сообщений - 26
В cl-gir есть пример, копипастю его сюда:


(defvar *gtk* (gir:ffi "Gtk"))

(defun main ()
  (gir:invoke (*gtk* 'init) nil)
  (let ((window (gir:invoke (*gtk* "Window" 'new)
			    (gir:nget *gtk* "WindowType" :toplevel)))
        (button (gir:invoke (*gtk* "Button" 'new-with-label) "Hello, world!")))
    (gir::g-signal-connect-data (gir::object-this window)
                                "destroy"
                                (cffi:foreign-symbol-pointer "gtk_main_quit")
                                (cffi:null-pointer)
                                (cffi:null-pointer)
                                0)
    (gir:connect button :clicked 
                 (lambda (button)
                   (setf (gir:property button 'label) "OK")))
    (gir:invoke (window 'add) button)
    (gir:invoke (window 'show-all))
    (gir:invoke (*gtk* 'main))))

Я вижу тут две привязки к сигналам destroy и clicked. Нельзя ли
destroy также связать через gir:connect? Можно ли получать доступ к
функциям gtk_main_quit через *gtk* и вызвать её?

У меня (gir:nget *gtk* 'main-quit) достаёт вроде что-то, но
(gir:invoke (*gtk* 'main-quit)) как-то непонятно работает.
[#]
> Нельзя ли destroy также связать через gir:connect? Можно ли получать доступ к функциям gtk_main_quit через *gtk* и вызвать её?

Можно. Здесь пример такой, чтобы продемонстрировать разные варианты привязки: низкий уровень через g-signal-connect-data и высокий через connect

Должно работать и так:

(defvar *gtk* (gir:ffi "Gtk"))

(defun main ()
  (gir:invoke (*gtk* 'init) nil)
  (let ((window (gir:invoke (*gtk* "Window" 'new)
                            (gir:nget *gtk* "WindowType" :toplevel)))
        (button (gir:invoke (*gtk* "Button" 'new-with-label) "Hello, world!")))
    (gir:connect window :destroy
                 (lambda (window)
                   (gir:invoke (*gtk* 'main-quit))))
    (gir:connect button :clicked
                 (lambda (button)
                   (setf (gir:property button 'label) "OK")))
    (gir:invoke (window 'add) button)
    (gir:invoke (window 'show-all))
    (gir:invoke (*gtk* 'main))))

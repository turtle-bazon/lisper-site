Ну смотри, допустим есть шаблон "(a|b)+s?", если метапрограммировать, то получится

string match-my-regex (string input)
{
    string result;

    while ( ! end-of-string (input) )
    {
        while ( first (input) == 'a'
             || first (input) == 'b')
        {
          result += first (input);
          ++ input;
        }

        if ( result . size != 0
          && first (input) == 's')
        {
           return (result += 's');
        }
        else
        {
           ++ input;
           result . clear ();
        }
    }

    return result;
}

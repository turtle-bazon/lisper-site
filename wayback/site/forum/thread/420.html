  (tv-nsec :long))

  (interval  :pointer)
  (remainder :pointer))

(defun nanosleep (seconds nano-seconds)
  (declare (type (unsigned-byte 32) seconds)
           (type (integer 0 1000000000) nano-seconds))
  (with-foreign-objects ((interval  'timespec)
                         (remainder 'timespec))
    (setf (foreign-slot-value interval  'timespec 'tv-sec)  seconds
          (foreign-slot-value interval  'timespec 'tv-nsec) nano-seconds)
    (%nanosleep interval remainder)))

А ещё (будет самое лучшее разрешение) можно засыпать на n-ое количество nop-ов


(sb-c:defknown %%nop () t)

(sb-c:define-vop (%%nop)
  (:translate %%nop)
  (:policy :fast-safe)
  (:generator 0 (sb-c::inst nop)))

(declaim (inline %nop))
(defun %nop () (%%nop))

(defmacro nop (count)
  `(progn
     ,@(loop :repeat count
             :collect '(%nop))))

Ещё одна попытка:

;;;;
;;;; sleep.lisp - portable *sleep functions.
;;;;
;;;;   Note: require `cygwin1.dll' on win32 platform.
;;;;

(defpackage #:sleep
  (:use     #:common-lisp
            #:cffi)
  (:export  #:usleep
            #:nanosleep
            #:millisleep))

(in-package #:sleep)

;;;; using `cygwin1.dll' on win32 platform

#+win32
(progn
  (define-foreign-library cygwin1
    (t (:default "cygwin1")))
  (use-foreign-library cygwin1))

;;;; foreign types

;;; From the `usleep' man page:
;;;
;;;   The type useconds_t is an unsigned integer type capable of holding integers in the
;;;   range [0, 1000000]. Programs will be more portable if they never mention this type
;;;   explicitly. Use the `unsigned int' type instead of `unsigned long' or `useconds_t'.
;;;
(defctype portable-useconds-t :unsigned-int)

(defctype time-t
  #-(or openbsd netbsd) :long ;; FIXME: include cygwin/win32 (?)
  #+(or openbsd netbsd) :int)

(defcstruct timespec
  (tv-sec  time-t)
  (tv-nsec :long))

;;;; foreign functions

(declaim (inline %usleep))
(defcfun ("usleep" %usleep) :void
  (usec portable-useconds-t))

(declaim (inline %nanosleep))
(defcfun ("nanosleep" %nanosleep) :int
  (interval  :pointer)
  (remainder :pointer))

#+win32
(progn
  (declaim (inline %millisleep))
  (defcfun ("Sleep@4" %millisleep) :void
    (milliseconds :unsigned-short)))

;;;; funtions for export

(defun usleep (useconds)
  (declare (type (integer 0 1000000) useconds))
  (%usleep useconds)
  (values))

(defun nanosleep (seconds nano-seconds)
  (declare (type (unsigned-byte 32) seconds)
           (type (integer 0 1000000000) nano-seconds))
  (with-foreign-objects ((interval  'timespec)
                         (remainder 'timespec))
    (setf (foreign-slot-value interval 'timespec 'tv-sec)  seconds
          (foreign-slot-value interval 'timespec 'tv-nsec) nano-seconds)
    (%nanosleep interval remainder)))

(defun millisleep (milliseconds)
  (declare (type (integer 0 1000) milliseconds))
  #-win32
  (%usleep (* milliseconds 1000))
  #+win32
  (%millisleep milliseconds)
  (values))

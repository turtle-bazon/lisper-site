Может у вас сеть плохо работает? У меня так было - вроде все сетевые программы работают а sbcl не мог найти tcp-протокол.

Следующий код должен работать независимо от ОС:

(require :sb-bsd-sockets)

(setf socket (sb-bsd-sockets:make-inet-socket :stream :tcp))
              ; вернёт объект сокет, потом по этому дескриптору нужно осуществить соединение

; тоже самое, inet-socket это класс
(setf socket (make-instance 'sb-bsd-sockets:inet-socket :type :stream :protocol :tcp)))

; теперь нужно выполнить соединение
(sb-bsd-sockets:socket-connect socket
    (sb-bsd-sockets:host-ent-address (sb-bsd-sockets:get-host-by-name "localhost"))
    80)
; должен работать сервер на 80 порту localhost
; можно соединиться с любой точкой, например wwww.google.com вместо localhost

(setf stream (sb-bsd-sockets:socket-make-stream socket
                  :input t
                  :output t
                  :buffering :full
                  :element-type :default
                  :external-format :iso-8859-1)

; теперь с потоком stream можно работать, читать и писать с помощью обычных format и read-line
; или более "правильно", с помощью tcp-print-raw + print и tcp-read-raw + read


А как это использовать, зависит от того, что вы хотите написать - клиент или сервер.

Сокеты лучше пользовать через usocket.
Кроссплатформенность потому что.
Вот пример:
(defun simple-server (&optional (port 0)) ;;0 - random port
  (let ((server (usocket:socket-listen "127.0.0.1" port
                                       :element-type 'base-char))
        (crlf (map 'string #'code-char '(13 10)))
        (ws   '(#\Space #\Tab #\Newline #\Return)))
    (format t "~&Server listening on port ~a~%"
            (usocket:get-local-port server))
    (unwind-protect
      (loop for connection = (usocket:socket-accept server)
            for connection-stream = (usocket:socket-stream connection)
            do (loop for line = (progn
                                  (format connection-stream
                                          "Write something('quit' - close connection):~a"
                                          crlf)
                                  (force-output connection-stream)
                                  (read-line connection-stream nil nil))
                     initially                     
                     (format t "~%Accepted connection~%")
                     while line
                     if (string-equal "quit" (string-trim ws line))
                     do (format connection-stream "kthxbye~a" crlf)
                        (force-output connection-stream)
                        (loop-finish)
                     else
                     do (format t "User wrote: ~a~&" line)
                     finally
                     (usocket:socket-close connection)
                     (write-line "Connection closed")))
      (usocket:socket-close server))))

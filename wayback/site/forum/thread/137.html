по сравнению с c-inline одно большое преимущество, можно сделать так:
(de harmonic (n)
    ((-double-) n)
    (let ((r 0))
      ((-double-) r)
      #{ { double i=0;
           while (i<$n){ i+=1.0; $r+=1/i; };
         } #} r))
вместо
(de harmonic (n)
        ((-double-) n)
        (let* ((z 0) (i 0))
         ((-double-) z i)
          (while (< i n) (incr i) (incr z (/ i))))).
